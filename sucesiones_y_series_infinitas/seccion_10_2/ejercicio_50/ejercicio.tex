\item Se tiene dado un triángulo rectángulo $ABC$ con $\angle A = \theta$ y $|AC| = b$. Se dibuja $CD$ perpendicular a $AB$, $DE$ se traza perpendicular a $BC$, $EF \perp AB$, y se continúa este proceso indefinidamente como se indica en la figura. Encuentre la longitud total de todas las perpendiculares $|CD| + |DE| + |EF| + |FG| + \dots$ en términos de $b$ y $\theta$.
    
    \begin{center}
    \begin{tikzpicture}[scale=1.5]
        \coordinate (B) at (0,0);
        \coordinate (C) at (4,0);
        \coordinate (A) at (4,2.5);

        \draw[thick] (B) node[below left] {$B$} -- (C) node[below] {$C$} -- (A) node[above right] {$A$} -- cycle;

        \draw (3.8,0) -- (3.8,0.2) -- (4,0.2);

        \draw (3.8, 2.1) arc (240:270:0.4); 
        \node at (3.75, 2.3) {$\theta$};
        \node[right] at (4, 1.25) {$b$};

        \coordinate (D) at ($(A)!(C)!(B)$);
        \draw (C) -- (D) node[above left] {$D$};
        \draw ($(D)!0.15cm!(B)$) -- ($($(D)!0.15cm!(B)$)!0.15cm!90:(D)$) -- ($(D)!0.15cm!(C)$);

        \coordinate (E) at ($(B)!(D)!(C)$);
        \draw (D) -- (E) node[below] {$E$};
        \draw ($(E)!0.15cm!(C)$) -- ($($(E)!0.15cm!(C)$)!0.15cm!90:(E)$) -- ($(E)!0.15cm!(D)$);

        \coordinate (F) at ($(A)!(E)!(B)$);
        \draw (E) -- (F) node[above left] {$F$};

        \coordinate (G) at ($(B)!(F)!(C)$);
        \draw (F) -- (G) node[below] {$G$};

        \coordinate (H) at ($(A)!(G)!(B)$);
        \draw (G) -- (H) node[above left] {$H$};

        \coordinate (I) at ($(B)!(H)!(C)$);
        \draw (H) -- (I);
        \coordinate (J) at ($(A)!(I)!(B)$);
        \draw (I) -- (J);
        \coordinate (K) at ($(B)!(J)!(C)$);
        \draw (J) -- (K);
        \coordinate (L) at ($(A)!(K)!(B)$);
        \draw (K) -- (L);
    \end{tikzpicture}
    \end{center}
