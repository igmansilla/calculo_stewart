\item Supóngase que $f(x) = \displaystyle\sum_{n=0}^\infty a_n x^n$ para todo $x$.
    \begin{enumerate}
        \item[(a)] Si $f$ es un función impar, demuestre que $a_0 = a_2 = a_4 = \dots = 0$.
        \item[(b)] Si $f$ es una función par, demuestre que $a_1 = a_3 = a_5 = \dots = 0$.
    \end{enumerate}
