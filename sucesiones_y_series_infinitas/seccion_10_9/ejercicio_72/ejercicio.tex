    \begin{enumerate}
        \item[(a)] Empezando por la serie geométrica $\displaystyle\sum_{n=0}^\infty x^n$, encuentre la suma de la serie $\displaystyle\sum_{n=1}^\infty n x^{n-1}$ para $|x| < 1$.
        \item[(b)] Encuentre las sumas de las series siguientes. \\
        (i) $\displaystyle\sum_{n=1}^\infty n x^n, \quad |x| < 1$ \qquad\qquad (ii) $\displaystyle\sum_{n=1}^\infty \frac{n}{2^n}$
        \item[(c)] Encuentre las sumas de las series siguientes. \\
        (i) $\displaystyle\sum_{n=2}^\infty n(n-1)x^n, \quad |x| < 1$ \\
        (ii) $\displaystyle\sum_{n=2}^\infty \frac{n^2 - n}{2^n}$ \\
        (iii) $\displaystyle\sum_{n=1}^\infty \frac{n^2}{2^n}$
    \end{enumerate}
