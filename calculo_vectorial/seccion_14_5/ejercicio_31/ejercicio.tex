\item Sea $f$ un campo escalar y $\mathbf{F}$ un campo vectorial. Determine si cada una de las expresiones siguientes es un campo escalar, un campo vectorial, o carece de sentido.
    \begin{enumerate}
        \item[(a)] $\operatorname{rot} f$
        \item[(b)] $\operatorname{grad} f$
        \item[(c)] $\operatorname{div} \mathbf{F}$
        \item[(d)] $\operatorname{rot}(\operatorname{grad} f)$
        \item[(e)] $\operatorname{grad} \mathbf{F}$
        \item[(f)] $\operatorname{grad}(\operatorname{div} \mathbf{F})$
        \item[(g)] $\operatorname{div}(\operatorname{grad} f)$
        \item[(h)] $\operatorname{grad}(\operatorname{div} f)$
        \item[(i)] $\operatorname{rot}(\operatorname{rot} \mathbf{F})$
        \item[(j)] $\operatorname{div}(\operatorname{div} \mathbf{F})$
        \item[(k)] $(\operatorname{grad} f) \times (\operatorname{div} \mathbf{F})$
        \item[(l)] $\operatorname{div}(\operatorname{rot}(\operatorname{grad} f))$
    \end{enumerate}
