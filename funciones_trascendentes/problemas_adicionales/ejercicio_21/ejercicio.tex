\item 
    \begin{enumerate}
        \item[(a)] Demuestre que $\ln x < x - 1$ para $x > 0, x \ne 1$.
        \item[(b)] Demuestre que para $x > 0, x \ne 1$,
        \[ \frac{x - 1}{x} < \ln x \]
        \item[(c)] Deduzca la desigualdad de Napier:
        \[ \frac{1}{a} < \frac{\ln a - \ln b}{a - b} < \frac{1}{b} \]
        si $a > b > 0$.
    \end{enumerate}
