\item Suponga que $\lim_{x \to a} f(x) = \infty$ y $\lim_{x \to a} g(x) = c$, en donde $c$ es un número real.
    \begin{enumerate}
        \item[(a)] Si $c > 0$, pruebe que $\lim_{x \to a} [f(x) g(x)] = \infty$.
        \item[(b)] Si $c < 0$, pruebe que $\lim_{x \to a} [f(x) g(x)] = -\infty$.
    \end{enumerate}
