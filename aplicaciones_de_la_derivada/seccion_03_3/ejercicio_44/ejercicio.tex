    \begin{enumerate}
        \item[(a)] $f'(4) = f'(10) = 0$
        \item[(b)] $f'(x) < 0 \text{ si } |x| < 2, \, f'(x) \ge 0 \text{ si } 2 < x < 10, \, f'(x) < 0 \text{ si } x > 10$
        \item[(c)] $f'(x) = 1 \text{ si } x < -2, \, \displaystyle \lim_{x \to -2^+} f'(x) = -1$
    \end{enumerate}
