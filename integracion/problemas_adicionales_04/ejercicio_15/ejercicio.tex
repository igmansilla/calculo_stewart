\item Suponga que $f$ y $g$ son funciones tales que $f(0) = 0$, $g(0) = 1$ y $f'(x) = g(x)$, $g'(x) = -f(x)$, para todo $x$.
\begin{enumerate}
    \item[(a)] Demuestre que $[f(x)]^2 + [g(x)]^2 = 1$ para todo $x$.
    \item[(b)] Demuestre que $[f(x) - \operatorname{sen} x]^2 + [g(x) - \cos x]^2 = 0$ para todo $x$.
    \item[(c)] Deduzca que $f(x) = \operatorname{sen} x$ y $g(x) = \cos x$ para todo $x$.
\end{enumerate}
