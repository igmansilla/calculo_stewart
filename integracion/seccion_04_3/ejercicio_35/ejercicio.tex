\item Determine cuáles de las siguientes funciones son integrables en el intervalo $[0, 2]$.
\begin{enumerate}
    \begin{minipage}{0.45\textwidth}
    \item[(a)] $f(x) = x^2 \operatorname{sen} x$
    \item[(c)] $f(x) = \begin{cases} x + 1 & \text{si } 0 \le x < 1 \\ 2 - x & \text{si } 1 \le x \le 2 \end{cases}$
    \end{minipage}%
    \begin{minipage}{0.45\textwidth}
    \item[(b)] $f(x) = \sec x$
    \item[(d)] $f(x) = \begin{cases} (x - 1)^{-2} & \text{si } x \ne 1 \\ 1 & \text{si } x = 1 \end{cases}$
    \end{minipage}
\end{enumerate}
