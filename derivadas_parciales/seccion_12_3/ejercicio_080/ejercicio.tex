\item Determine cuáles de las siguientes funciones son soluciones de la ecuación de Laplace $u_{xx} + u_{yy} = 0$.
\begin{itemize}
    \item[(a)] $u = x^2 + y^2$
    \item[(b)] $u = x^2 - y^2$
    \item[(c)] $u = x^3 + 3xy^2$
    \item[(d)] $u = \ln\sqrt{x^2+y^2}$
    \item[(e)] $u = \operatorname{sen} x \cosh y + \cos x \operatorname{senh} y$
    \item[(f)] $u = e^{-x} \cos y - e^{-y} \cos x$
\end{itemize}
