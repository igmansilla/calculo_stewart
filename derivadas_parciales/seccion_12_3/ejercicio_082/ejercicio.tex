\item Demuestre que las siguientes funciones son soluciones de la ecuación de onda $u_{tt} = a^2 u_{xx}$.
\begin{itemize}
    \item[(a)] $u = \operatorname{sen}(kx) \operatorname{sen}(akt)$
    \item[(b)] $u = t / (a^2t^2 - x^2)$
    \item[(c)] $u = (x-at)^6 + (x+at)^6$
    \item[(d)] $u = \operatorname{sen}(x-at) + \ln(x+at)$
\end{itemize}
