% Sección 5.1: Áreas entre Curvas

Representación analítica y geométrica de áreas de regiones planas delimitadas por curvas algebraicas y trascendentes, integrando respecto a $x$ y respecto a $y$.

\begin{center}
\begin{tikzpicture}[scale=1.2]
    % Ejes
    \draw[->] (-0.5,0) -- (6,0) node[right] {$x$};
    \draw[->] (0,-1.5) -- (0,1.5) node[above] {$y$};
    
    % Curva
    \draw[thick] plot [smooth, tension=0.7] coordinates {(0,0.8) (0.8, 1.2) (2,0) (3,-1) (4,0) (5, 0.6)};
    \node[right] at (5, 0.6) {$y = f(x)$};
    
    % Etiquetas en los ejes
    \node[below left] at (0,0) {$0$};
    \draw (0.2, 0.1) -- (0.2, -0.1) node[below] {$a$};
    \draw (2, 0.1) -- (2, -0.1) node[above left] {$c$};
    \draw (4, 0.1) -- (4, -0.1) node[above left] {$d$};
    \draw (5.2, 0.1) -- (5.2, -0.1) node[below] {$b$};
    
    % Signos de las áreas
    \node at (1.1, 0.5) {$+$};
    \node at (3, -0.5) {$-$};
    \node at (4.6, 0.25) {$+$};
\end{tikzpicture}

\end{center}

\begin{align*}
\int_a^b f &= \int_a^c f + \int_c^d f + \int_d^b f \\
&= \left( \int_a^c f + \int_d^b f \right) - \left( -\int_c^d f \right) \\
&= A_1 - A_2
\end{align*}
