$$\lim_{x \to 0} \frac{f(x)}{x} = 4 \quad \lim_{x \to 0} \frac{g(x)}{x} = 2$$
\begin{enumerate}
    \begin{minipage}{0.3\textwidth}

    \end{minipage}
    \begin{minipage}{0.3\textwidth}
    \item[(b)] Encuentre $f'(0)$.
    \end{minipage}
    \begin{minipage}{0.3\textwidth}
    \item[(c)] Encuentre $\displaystyle \lim_{x \to 0} \frac{g(x)}{f(x)}$.
    \end{minipage}
\end{enumerate}
