    \begin{enumerate}
        \item[(a)] Use una sustitución trigonométrica para demostrar que
        \[ \int \frac{dx}{\sqrt{x^2 + a^2}} = \ln(x + \sqrt{x^2 + a^2}) + C \]
        \item[(b)] Utilice la sustitución hiperbólica $x = a \senh t$ para demostrar que

    \end{enumerate}
