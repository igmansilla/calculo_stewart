\item 
    \begin{enumerate}
        \item[(a)] Demuestre que $\int_{-\infty}^{\infty} x \, dx$ es divergente.
        \item[(b)] Demuestre que
        \[ \lim_{t \to \infty} \int_{-t}^{t} x \, dx = 0 \]
        Esto prueba que no se puede definir
        \[ \int_{-\infty}^{\infty} f(x) \, dx = \lim_{t \to \infty} \int_{-t}^{t} f(x) \, dx \]
    \end{enumerate}
